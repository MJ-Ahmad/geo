  <script type="module">
    // Load all GeoGrid data
    const bangladeshHierarchy = await fetch('/modules/geogrid/data/bangladesh-hierarchy.json').then(r => r.json());
    const controlHierarchy = await fetch('/modules/geogrid/data/control-hierarchy.json').then(r => r.json());
    const routesSystem = await fetch('/modules/geogrid/data/routes-system.json').then(r => r.json());
    const surveillanceZones = await fetch('/modules/geogrid/data/surveillance-zones.json').then(r => r.json());

    // Initialize map
    const map = L.map('map').setView([23.8, 90.4], 7);
    L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
      attribution: '© OpenStreetMap'
    }).addTo(map);

    // Render divisions
    bangladeshHierarchy.divisions.forEach(division => {
      L.circleMarker([division.coordinates[0], division.coordinates[1]], {
        radius: 12,
        color: '#47d7b3',
        fillColor: '#47d7b3',
        fillOpacity: 0.9,
        weight: 2
      }).bindPopup(`<strong>${division.name_en}</strong><br/>Population: ${division.population.toLocaleString()}`).addTo(map);
    });

    // Render surveillance zones
    surveillanceZones.surveillance_zones.forEach(zone => {
      L.circle([zone.coordinates[0], zone.coordinates[1]], {
        radius: zone.radius_km * 1000,
        color: '#79b7ff',
        fillColor: '#79b7ff',
        fillOpacity: 0.08,
        weight: 1.5
      }).bindPopup(`<strong>${zone.name_en}</strong><br/>Security: ${zone.security_level}`).addTo(map);
    });

    // Render routes
    routesSystem.land_routes.national_highways.forEach(route => {
      L.polyline(route.route, {
        color: '#e7b75a',
        weight: 3,
        opacity: 0.8
      }).bindPopup(`<strong>${route.name_en}</strong><br/>Length: ${route.length_km} km`).addTo(map);
    });

    // Populate UI
    document.getElementById('hierarchy-list').innerHTML = bangladeshHierarchy.divisions.slice(0, 4).map(d => `
      <div class="item">
        <span>${d.name_en}</span>
        <span class="badge">${d.total_districts || d.districts.length} districts</span>
      </div>
    `).join('');

    document.getElementById('routes-list').innerHTML = `
      <div class="item"><span>Land Routes</span><span class="badge">Active</span></div>
      <div class="item"><span>Water Routes</span><span class="badge">Active</span></div>
      <div class="item"><span>Air Corridors</span><span class="badge">Active</span></div>
    `;

    document.getElementById('monitoring-stats').innerHTML = `
      <div class="stat-box">
        <div class="stat-label">Total Coverage</div>
        <div class="stat-value">${surveillanceZones.risk_assessment.coverage_percentage}%</div>
      </div>
      <div class="stat-box">
        <div class="stat-label">Active Zones</div>

        <div class="stat-label">Incidents (24h)</div>
        <div class="stat-value">${surveillanceZones.surveillance_zones.reduce((sum, z) => sum + z.incidents_24h, 0)}</div>
      </div>
    `;
