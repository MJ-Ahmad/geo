```md
# Geo-Grid Control System

This module is the operational geographic backbone of the National Workforce Grid.

## Purpose
- National geographic intelligence
- Zone monitoring and command
- Land, water, and air route control
- Decision escalation and response workflow
- Policy-driven surveillance and coordination

## Hierarchy
- Division
- District
- Thana / Police Station
- Union
- Ward
- Neighborhood

## Command Flow
Neighborhood → Ward → Union → Thana → District → Division → National

## Design Principles
- Single national control model
- Role-based authority
- Policy-controlled escalation
- Real-time observation and response
- Bangladesh-specific geography and operational hierarchy

## File Architecture
- data/hierarchy.json
- data/routes.json
- data/surveillance.json
- data/decision-flow.json
- js/map-engine.js
- js/monitoring-engine.js
- js/decision-engine.js
```

```javascript
export class GeoGridMapEngine {
  constructor(containerId) {
    this.map = L.map(containerId).setView([23.8, 90.4], 7);
    this.baseLayer = L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
      attribution: '&copy; OpenStreetMap contributors'
    }).addTo(this.map);
  }

  renderHierarchy(divisions) {
    divisions.forEach(div => {
      L.circleMarker([div.coordinates[0], div.coordinates[1]], {
        radius: 11,
        color: '#47d7b3',
        fillColor: '#47d7b3',
        fillOpacity: 0.9,
        weight: 2
      }).bindPopup(`<strong>${div.name_en}</strong><br/>Districts: ${div.district_count || 0}`).addTo(this.map);
    });
  }

  renderSurveillance(zones) {
    zones.forEach(zone => {
      L.circle([zone.coordinates[0], zone.coordinates[1]], {
        radius: zone.radius_km * 1000,
        color: '#79b7ff',
        fillColor: '#79b7ff',
        fillOpacity: 0.08,
        weight: 1.5
      }).bindPopup(`<strong>${zone.name_en}</strong><br/>Security: ${zone.security_level}<br/>Incidents: ${zone.incidents_24h}`).addTo(this.map);
    });
  }

  renderRoutes(routes) {
    routes.land.forEach(route => {
      L.polyline(route.route, { color: '#e7b75a', weight: 3, opacity: 0.9 }).addTo(this.map);
    });

    routes.water.forEach(route => {
      L.polyline(route.route, { color: '#79b7ff', weight: 2.5, opacity: 0.8, dashArray: '10,10' }).addTo(this.map);
    });

    routes.air.forEach(route => {
      L.polyline(route.route, { color: '#ff6b6b', weight: 2.2, opacity: 0.8, dashArray: '6,8' }).addTo(this.map);
    });
  }
}
```

```javascript
export class MonitoringEngine {
  constructor() {
    this.incidents = [];
  }

  fromSurveillance(zones) {
    this.incidents = zones.map(zone => ({
      id: zone.id,
      name: zone.name_en,
      incidents_24h: zone.incidents_24h,
      security_level: zone.security_level,
      response_time_sec: zone.response_time_sec
    }));
    return this.incidents;
  }

  renderIncidentList(containerId) {
    const el = document.getElementById(containerId);
    if (!el) return;

    el.innerHTML = this.incidents.map(item => `
      <div class="mini-item">
        <span>${item.name}</span>
        <strong>${item.incidents_24h}</strong>
      </div>
    `).join('');
  }
}
```
